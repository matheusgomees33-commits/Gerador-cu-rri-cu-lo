//   ABAS

const ABAS = ['pessoal', 'experiencia', 'educacao', 'extras'];
let abaAtual = ABAS[0];

function initAbas() {
  document.querySelectorAll('.aba').forEach(btn => {
    btn.addEventListener('click', () => ativarAba(btn.dataset.aba));
  });
  ativarAba(abaAtual);
}

function ativarAba(nome) {
  if (!ABAS.includes(nome)) return;
  abaAtual = nome;
  document.querySelectorAll('.aba').forEach(b => {
    const ativa = b.dataset.aba === nome;
    b.classList.toggle('aba--ativa', ativa);
    b.setAttribute('aria-selected', String(ativa));
  });
  document.querySelectorAll('.aba-conteudo').forEach(c =>
    c.classList.toggle('aba-conteudo--ativa', c.id === 'aba-' + nome));
  const i = ABAS.indexOf(nome);
  const ant = document.querySelector('[data-acao="anterior"]');
  const prox = document.querySelector('[data-acao="proximo"]');
  if (ant)  ant.disabled  = i === 0;
  if (prox) prox.disabled = i === ABAS.length - 1;
}

// Botões gerados por tools/gerar_botoes.py usam data-acao; aqui fica o comportamento.
const ACOES = {
  anterior: () => ativarAba(ABAS[ABAS.indexOf(abaAtual) - 1]),
  proximo:  () => ativarAba(ABAS[ABAS.indexOf(abaAtual) + 1]),
  limpar:   () => limparFormulario(),
  pdf:      () => exportarPDF(),
  html:     () => exportarCV()
};

function initAcoes() {
  document.addEventListener('click', e => {
    const btn = e.target.closest('[data-acao]');
    if (btn && !btn.disabled) ACOES[btn.dataset.acao]?.();
  });
}

function atualizarProgresso(d) {
  const itens = [d.nome, d.cargo, d.email, d.resumo,
                 d.experiencias.length, d.educacoes.length, d.habilidades.length];
  const pct = Math.round(itens.filter(Boolean).length / itens.length * 100);
  const barra = document.getElementById('progresso-barra');
  const txt   = document.getElementById('progresso-texto');
  if (barra) { barra.style.width = pct + '%'; barra.setAttribute('aria-valuenow', pct); }
  if (txt)   txt.textContent = pct + '% completo';
}

function coletarDados() {
  return {
    nome:        obter('nome'),
    cargo:       obter('cargo'),
    email:       obter('email'),
    telefone:    obter('telefone'),
    cidade:      obter('cidade'),
    linkedin:    obter('linkedin'),
    github:      obter('github'),
    instagram:   obter('instagram'),
    resumo:      obter('resumo'),
    habilidades: obter('habilidades').split(',').map(s => s.trim()).filter(Boolean),
    idiomas:     obter('idiomas').split(',').map(s => s.trim()).filter(Boolean),
    cursos:      obter('cursos').split('\n').map(s => s.trim()).filter(Boolean),
    mensagem:    obter('mensagem'),
    experiencias,
    educacoes
  };
}

function gerarHTMLCV(d) {
  const contatos = [];
  if (d.email)     contatos.push({ texto: d.email,     href: 'mailto:' + d.email });
  if (d.telefone)  contatos.push({ texto: d.telefone,  href: 'tel:' + d.telefone.replace(/\D/g, '') });
  if (d.cidade)    contatos.push({ texto: d.cidade,    href: null });
  if (d.linkedin)  contatos.push({ texto: d.linkedin,  href: 'https://' + d.linkedin.replace(/^https?:\/\//, '') });
  if (d.github)    contatos.push({ texto: d.github,    href: 'https://' + d.github.replace(/^https?:\/\//, '') });
  if (d.instagram) contatos.push({ texto: d.instagram, href: 'https://' + d.instagram.replace(/^https?:\/\//, '') });

  let html = '';

  // CABEÇALHO
  html += `
  <div class="cv__cabecalho">
    <div>
      <div class="cv__nome">${esc(d.nome) || 'Seu Nome Aqui'}</div>
      ${d.cargo ? `<div class="cv__cargo">${esc(d.cargo)}</div>` : ''}
    </div>
    <div class="cv__contatos">
      ${contatos.map(c =>
        `<div class="cv__contato">${
          c.href ? `<a href="${esc(c.href)}">${esc(c.texto)}</a>` : esc(c.texto)
        }</div>`
      ).join('')}
    </div>
  </div>`;

  if (d.resumo) html += secao('Sobre mim', `<div class="cv__resumo">${esc(d.resumo)}</div>`);

  if (d.experiencias.length > 0) {
    const corpo = d.experiencias.map(exp => `
      <div class="cv__exp">
        <div class="cv__exp-topo">
          <span class="cv__exp-empresa">${esc(exp.empresa) || 'Empresa'}</span>
          ${exp.periodo ? `<span class="cv__exp-periodo">${esc(exp.periodo)}</span>` : ''}
        </div>
        ${exp.cargo    ? `<div class="cv__exp-cargo">${esc(exp.cargo)}</div>` : ''}
        ${exp.descricao ? `<div class="cv__exp-desc">${esc(exp.descricao)}</div>` : ''}
      </div>
    `).join('');
    html += secao('Experiências profissionais', corpo);
  }

  if (d.educacoes.length > 0) {
    const corpo = d.educacoes.map(edu => `
      <div class="cv__edu">
        <div class="cv__edu-curso">${esc(edu.curso) || 'Curso'}</div>
        ${edu.instituicao ? `<div class="cv__edu-inst">${esc(edu.instituicao)}</div>` : ''}
        ${edu.periodo     ? `<div class="cv__edu-periodo">${esc(edu.periodo)}</div>` : ''}
      </div>
    `).join('');
    html += secao('Formação acadêmica', corpo);
  }

  if (d.cursos.length > 0) {
    html += secao('Cursos e certificados',
      `<div class="cv__cursos">${
        d.cursos.map(c => `<span class="cv__curso">${esc(c)}</span>`).join('')
      }</div>`
    );
  }

  const temHab = d.habilidades.length > 0;
  const temIdi = d.idiomas.length > 0;

  if (temHab || temIdi) {
    let cols = '';
    if (temHab) {
      cols += `
        <div class="cv__secao">
          <div class="cv__secao-titulo">Habilidades técnicas</div>
          <div class="cv__habilidades">
            ${d.habilidades.map(h => `<span class="cv__habilidade">${esc(h)}</span>`).join('')}
          </div>
        </div>`;
    }
    if (temIdi) {
      cols += `
        <div class="cv__secao">
          <div class="cv__secao-titulo">Idiomas</div>
          ${d.idiomas.map(i => `<div class="cv__idioma">${esc(i)}</div>`).join('')}
        </div>`;
    }
    html += `<div class="cv__duas-col">${cols}</div>`;
  }

  if (d.mensagem) {
    html += secao('Por que devo ser contratado?',
      `<div class="cv__mensagem">${esc(d.mensagem)}</div>`
    );
  }

  html += `
  <div class="cv__rodape">
    ${esc(d.nome) || 'Currículo'} · Gerado em ${new Date().toLocaleDateString('pt-BR')} · ADS 2026
  </div>`;

  return html;
}

function exportarCV() {
  const d = coletarDados();
  if (!d.nome) {
    alert('Preencha pelo menos o campo "Nome completo" antes de exportar!');
    return;
  }

  const arquivo = `<!DOCTYPE html>
<html lang="pt-BR">
<head>
  <meta charset="UTF-8" />
  <meta name="viewport" content="width=device-width, initial-scale=1.0" />
  <title>Currículo — ${esc(d.nome)}</title>
  <link href="https://fonts.googleapis.com/css2?family=Playfair+Display:wght@700&family=Inter:wght@400;500;600&display=swap" rel="stylesheet" />
  <style>${getCVInlineCSS()}</style>
</head>
<body>
  <div class="cv">
    ${gerarHTMLCV(d)}
  </div>
</body>
</html>`;

  const blob = new Blob([arquivo], { type: 'text/html;charset=utf-8' });
  const url  = URL.createObjectURL(blob);
  const link = document.createElement('a');
  const nome = normalizarNome(d.nome);

  link.href     = url;
  link.download = 'curriculo-' + nome + '.html';
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
  URL.revokeObjectURL(url);
}

// ================================================
//   EXPORTAR PDF (html2pdf.js)
// ================================================
async function exportarPDF() {
  if (typeof html2pdf === 'undefined') {
    alert('Biblioteca de PDF ainda carregando. Aguarde alguns segundos e tente novamente.');
    return;
  }

  const d = coletarDados();
  if (!d.nome) {
    alert('Preencha pelo menos o campo "Nome completo" antes de exportar!');
    return;
  }

  const btnPDF = document.getElementById('btn-exportar-pdf');
  const textoOriginal = btnPDF ? btnPDF.textContent : '';
  if (btnPDF) {
    btnPDF.disabled    = true;
    btnPDF.textContent = 'Gerando PDF...';
  }

  // Cria um container oculto com os estilos inline e o HTML do CV
  const container = document.createElement('div');
  container.style.cssText = 'position:fixed;left:-9999px;top:0;width:794px;background:#fff;padding:0;z-index:-1;';

  const styleEl = document.createElement('style');
  styleEl.textContent = getCVInlineCSS();
  container.appendChild(styleEl);

  const cvDiv = document.createElement('div');
  cvDiv.className = 'cv';
  cvDiv.style.cssText = 'padding:44px 48px;max-width:100%;width:100%;box-shadow:none;border-radius:0;';
  cvDiv.innerHTML = gerarHTMLCV(d);
  container.appendChild(cvDiv);

  document.body.appendChild(container);

  const nome = normalizarNome(d.nome);

  try {
    await html2pdf().set({
      margin:      [8, 10, 8, 10],
      filename:    'curriculo-' + nome + '.pdf',
      image:       { type: 'jpeg', quality: 0.97 },
      html2canvas: { scale: 2, useCORS: true, logging: false, letterRendering: true },
      jsPDF:       { unit: 'mm', format: 'a4', orientation: 'portrait' }
    }).from(cvDiv).save();
  } catch (err) {
    console.error('Erro ao gerar PDF:', err);
    alert('Erro ao gerar PDF. Alternativa: use Ctrl+P e salve como PDF pelo navegador.');
  } finally {
    document.body.removeChild(container);
    if (btnPDF) {
      btnPDF.disabled    = false;
      btnPDF.textContent = textoOriginal;
    }
  }
}

function montarSnapshot() {
  const campos = {};
  CAMPOS.forEach(id => {
    const el = document.getElementById(id);
    campos[id] = el ? el.value : '';
  });
  return {
    versao: 1,
    campos,
    experiencias: experiencias.map(({ empresa, cargo, periodo, descricao }) =>
      ({ empresa, cargo, periodo, descricao })),
    educacoes: educacoes.map(({ curso, instituicao, periodo }) =>
      ({ curso, instituicao, periodo }))
  };
}

function agendarSalvamento() {
  if (restaurando) return;
  salvamentoPendente = true;
  mostrarStatus('salvando');
  clearTimeout(timerSalvar);
  timerSalvar = setTimeout(salvarAgora, SALVAR_DELAY_MS);
}

function salvarSePendente() {
  if (salvamentoPendente) salvarAgora();
}

function salvarAgora() {
  clearTimeout(timerSalvar);
  timerSalvar = null;
  salvamentoPendente = false;
  try {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(montarSnapshot()));
    mostrarStatus('salvo');
    return true;
  } catch (err) {
    console.error('Erro ao salvar currículo:', err);
    mostrarStatus('erro', 'Não foi possível salvar: armazenamento do navegador indisponível ou cheio.');
    return false;
  }
}

// Retorna true se restaurou dados salvos; false se não há dados (ou são inválidos): o formulário fica em branco.
// Valida e normaliza tudo ANTES de alterar a tela, e gera novos ids para os itens
// (ids vão para atributos onclick e não devem vir do storage).
function carregarSalvo() {
  let bruto;
  try {
    bruto = localStorage.getItem(STORAGE_KEY);
  } catch (err) {
    console.error('localStorage indisponível:', err);
    mostrarStatus('erro', 'Armazenamento do navegador indisponível: suas alterações não serão salvas.');
    return false;
  }
  if (bruto === null) return false;

  const texto = v => (typeof v === 'string' ? v : '');
  const lista = v => (Array.isArray(v) ? v.filter(x => x && typeof x === 'object') : []);

  let dados;
  try {
    const s = JSON.parse(bruto);
    if (!s || typeof s !== 'object' || !s.campos || typeof s.campos !== 'object') {
      throw new Error('formato inválido');
    }
    dados = {
      campos: Object.fromEntries(CAMPOS.map(id => [id, texto(s.campos[id])])),
      exps:   lista(s.experiencias).map(e => [texto(e.empresa), texto(e.cargo), texto(e.periodo), texto(e.descricao)]),
      edus:   lista(s.educacoes).map(e => [texto(e.curso), texto(e.instituicao), texto(e.periodo)])
    };
  } catch (err) {
    console.warn('Dados salvos ilegíveis; iniciando em branco.', err);
    mostrarStatus('erro', 'Os dados salvos estavam ilegíveis e foram ignorados.');
    return false;
  }

  CAMPOS.forEach(id => definir(id, dados.campos[id]));
  experiencias = [];
  educacoes    = [];
  contExp = 0;
  contEdu = 0;
  dados.exps.forEach(args => adicionarExperiencia(...args));
  dados.edus.forEach(args => adicionarEducacao(...args));
  return true;
}

function mostrarStatus(estado, mensagem = '') {
  const el = document.getElementById('status-salvamento');
  if (!el) return;
  const hora = new Date().toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit' });
  const textos = {
    salvando: 'Salvando…',
    salvo:    `✓ Salvo neste navegador às ${hora}`,
    erro:     mensagem
  };
  el.dataset.estado = estado;
  el.textContent    = textos[estado];
  el.hidden         = false;
}

// ================================================
//   UTILITÁRIOS
// ================================================
function esc(str) {
  return (str || '')
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;');
}

function normalizarNome(nome) {
  return (nome || 'curriculo').toLowerCase()
    .normalize('NFD').replace(/[\u0300-\u036f]/g, '')
    .replace(/\s+/g, '-').replace(/[^a-z0-9-]/g, '');
}

function adicionarVaga(titulo = '', area = '', requisitos = '', descricao = '', nivel = '') {
  contVaga++;
  vagas.push({
    id: 'vaga-' + contVaga,
    titulo,
    area,
    requisitos,
    descricao,
    nivel
  });
  renderizarVagas();
}

function removerVaga(id) {
  if (vagas.length <= 1) {
    mostrarToast('Mantenha pelo menos uma vaga cadastrada.', 'erro');
    return;
  }
  vagas = vagas.filter(v => v.id !== id);
  renderizarVagas();
}

function atualizarVaga(id, campo, valor) {
  const item = vagas.find(v => v.id === id);
  if (item) item[campo] = valor;
}

function renderizarVagas() {
  const container = document.getElementById('lista-vagas');
  if (!container) return;
  container.innerHTML = '';

  vagas.forEach((vaga, i) => {

    card.className = 'vaga-card';
    card.innerHTML = `
      <div class="vaga-card__topo">
        <span class="vaga-card__num">Vaga ${i + 1}</span>
        <button class="vaga-card__rem" onclick="removerVaga('${vaga.id}')">Remover</button>
      </div>
      <div class="campo-linha">
        <div class="campo">
          <label>Título da vaga <span class="obrig">*</span></label>
          <input type="text" value="${esc(vaga.titulo)}"
            oninput="atualizarVaga('${vaga.id}', 'titulo', this.value)"
            placeholder="Ex: Desenvolvedor Front-End" />
        </div>
        <div class="campo">
          <label>Área / Departamento</label>
          <input type="text" value="${esc(vaga.area)}"
            oninput="atualizarVaga('${vaga.id}', 'area', this.value)"
            placeholder="Ex: Tecnologia, RH, Financeiro" />

//   PASSO 1 → 2: GERAR REQUISITOS TÉCNICOS

function gerarRequisitos() {
  const nomeEmpresa  = document.getElementById('nome-empresa')?.value.trim() || 'a empresa';
  const setorEmpresa = document.getElementById('setor-empresa')?.value.trim() || '';

  const vagasValidas = vagas.filter(v => v.titulo.trim());
  if (vagasValidas.length === 0) {
    mostrarToast('Preencha o título de pelo menos uma vaga antes de continuar.', 'erro');
    return;
  }

  const container = document.getElementById('lista-prompts');
  if (!container) return;
  container.innerHTML = '';

  // Limpa requisitos anteriores
  vagaReqTexto = {};

  vagasValidas.forEach((vaga, i) => {
    const textoReq = gerarTextoRequisitos(vaga, nomeEmpresa, setorEmpresa);
    // Guarda para uso no passo 3
    vagaReqTexto[vaga.id] = textoReq;

    const card = document.createElement('div');
    card.className = 'prompt-card';
    card.innerHTML = `
      <div class="prompt-card__topo">
        <span class="prompt-card__titulo">📄 Vaga ${i + 1}: ${esc(vaga.titulo)}${vaga.nivel ? ' — ' + esc(vaga.nivel) : ''}</span>
        <div class="prompt-card__acoes">
          <button class="btn-copiar" onclick="copiarPrompt(this, 'req-txt-${vaga.id}')">📋 Copiar</button>
        </div>
      </div>
      <textarea
        id="req-txt-${vaga.id}"
        class="prompt-card__textarea"
        rows="10"
        spellcheck="false"
        readonly
      >${esc(textoReq)}</textarea>
    `;
    container.appendChild(card);
  });

  // Popular select da triagem (passo 3)
  const select = document.getElementById('select-vaga-analise');
  if (select) {
    select.innerHTML = '<option value="">Selecione uma vaga...</option>';
    vagasValidas.forEach((vaga, i) => {
      const opt = document.createElement('option');
      opt.value = vaga.id;
      opt.textContent = `Vaga ${i + 1}: ${vaga.titulo}${vaga.nivel ? ' — ' + vaga.nivel : ''}`;
      select.appendChild(opt);
    });
  }

  // Mostrar passo 2
  document.getElementById('passo-1').style.display = 'none';
  const p2 = document.getElementById('passo-2');
  p2.style.display = 'block';
  setTimeout(() => p2.scrollIntoView({ behavior: 'smooth', block: 'start' }), 80);
}

// ================================================
//   GERAR TEXTO DE REQUISITOS TÉCNICOS
// ================================================
function gerarTextoRequisitos(vaga, empresa, setor) {
  const setorTexto = setor ? ` | Setor: ${setor}` : '';
  const nivel      = vaga.nivel     ? vaga.nivel     : 'Não especificado';
  const area       = vaga.area      ? vaga.area      : 'Não especificada';
  const data       = new Date().toLocaleDateString('pt-BR');

  // Requisitos como lista com bullet
  const reqLista = vaga.requisitos
    ? vaga.requisitos
        .split(',')
        .map(r => r.trim())
        .filter(Boolean)
        .map(r => `  • ${r}`)
        .join('\n')
    : '  • (Não informado)';

  const descricao = vaga.descricao
    ? vaga.descricao.trim()
    : '(Não informada)';

  return `╔══════════════════════════════════════════════════════╗
  REQUISITOS TÉCNICOS DA VAGA
  ${empresa}${setorTexto}
  Gerado em: ${data} | Tech-ZL · ADS 2026
╚══════════════════════════════════════════════════════╝

▶ CARGO          : ${vaga.titulo}
▶ NÍVEL          : ${nivel}
▶ ÁREA           : ${area}

━━━ REQUISITOS TÉCNICOS OBRIGATÓRIOS ━━━
${reqLista}

━━━ RESPONSABILIDADES E ATIVIDADES ━━━
${descricao}

━━━ CRITÉRIOS DE AVALIAÇÃO ━━━
  ✅ APROVADO       — Atende ≥ 70% dos requisitos acima
  ⚠️ EM ANÁLISE     — Atende entre 40% e 69% dos requisitos
  ❌ NÃO RECOMENDADO — Atende menos de 40% dos requisitos

━━━ INSTRUÇÕES PARA O AGENTE GEMS ━━━
Para cada candidato analisado, forneça:
  1. Nome do candidato
  2. Pontuação de 0 a 10
  3. Status: APROVADO | EM ANÁLISE | NÃO RECOMENDADO
  4. Pontos fortes em relação a esta vaga
  5. Pontos de atenção / lacunas
  6. Recomendação final objetiva`;
}

function mostrarRequisitosSelecionados() {
  const vagaId  = document.getElementById('select-vaga-analise')?.value;
  const bloco   = document.getElementById('bloco-requisitos-preview');
  const preview = document.getElementById('requisitos-preview');

  if (!vagaId || !bloco || !preview) {
    if (bloco) bloco.style.display = 'none';
    return;
  }

  const texto = vagaReqTexto[vagaId];
  if (texto) {
    preview.value      = texto;
    bloco.style.display = 'block';
  } else {
    bloco.style.display = 'none';
  }

  // Limpar resultado anterior ao trocar de vaga
  const resultado = document.getElementById('resultado-analise');
  if (resultado) resultado.innerHTML = '';
}

// ================================================
//   PASSO 3: PREPARAR MENSAGEM PARA O GEMS
//   — O site envia apenas os requisitos.
//   — O agente Gems solicitará os currículos no chat.
// ================================================
function prepararParaGemini() {
  const vagaId = document.getElementById('select-vaga-analise')?.value;

  if (!vagaId) {
    mostrarToast('Selecione uma vaga para o processo seletivo.', 'erro');
    return;
  }

  const vaga = vagas.find(v => v.id === vagaId);
  if (!vaga) {
    mostrarToast('Vaga não encontrada. Volte ao Passo 1.', 'erro');
    return;
  }

  const nomeEmpresa  = document.getElementById('nome-empresa')?.value.trim() || 'a empresa';
  const setorEmpresa = document.getElementById('setor-empresa')?.value.trim() || '';
  const requisitos   = vagaReqTexto[vagaId] || gerarTextoRequisitos(vaga, nomeEmpresa, setorEmpresa);

  const mensagem = gerarMensagemGemini(vaga, nomeEmpresa, setorEmpresa, requisitos);
  renderizarMensagemGemini(mensagem, vaga);
}

// ================================================
//   GERAR MENSAGEM DE REQUISITOS PARA O GEMS
//   — Apenas os requisitos técnicos da vaga.
//   — O agente pedirá os currículos no chat.
// ================================================
function gerarMensagemGemini(vaga, empresa, setor, requisitos) {
  const setorTexto = setor ? ` (${setor})` : '';
  return `[INÍCIO DO PROCESSO SELETIVO — ${vaga.titulo.toUpperCase()} | ${empresa}${setorTexto}]

${requisitos}

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Esses são os requisitos técnicos da vaga acima.
Por favor, confirme o recebimento e solicite os currículos dos candidatos para iniciar a triagem.`;
}

// ================================================
//   COPIAR PROMPT
// ================================================
function copiarPrompt(btn, textareaId) {
  const ta = document.getElementById(textareaId);
  if (!ta) return;

  navigator.clipboard.writeText(ta.value).then(() => {
    btn.textContent = '✅ Copiado!';
    btn.classList.add('copiado');
    setTimeout(() => {
      btn.textContent = '📋 Copiar';
      btn.classList.remove('copiado');
    }, 2000);
  }).catch(() => {
    // Fallback para browsers antigos
    ta.select();
    document.execCommand('copy');
    btn.textContent = '✅ Copiado!';
    setTimeout(() => { btn.textContent = '📋 Copiar'; }, 2000);
  });
}

// ================================================
//   LIMPAR TUDO
// ================================================
function limparTudo() {
  if (!confirm('Tem certeza que deseja limpar todos os dados?')) return;

  vagas        = [];
  contVaga     = 0;
  vagaReqTexto = {};

  const nomeEmp  = document.getElementById('nome-empresa');
  const setorEmp = document.getElementById('setor-empresa');
  if (nomeEmp)  nomeEmp.value  = '';
  if (setorEmp) setorEmp.value = '';

  const resultadoDiv = document.getElementById('resultado-analise');
  if (resultadoDiv) resultadoDiv.innerHTML = '';

  // Voltar ao passo 1
  document.getElementById('passo-2').style.display = 'none';
  document.getElementById('passo-3').style.display = 'none';
  document.getElementById('passo-1').style.display = 'block';

  adicionarVaga();
  mostrarToast('Dados limpos com sucesso.', 'ok');
}

// ================================================
//   TOAST NOTIFICATIONS
// ================================================
function mostrarToast(msg, tipo = 'ok') {
  document.querySelector('.toast-notif')?.remove();

  const toast = document.createElement('div');
  toast.className = 'toast-notif';
  toast.textContent = msg;

  Object.assign(toast.style, {
    position:     'fixed',
    bottom:       '28px',
    right:        '24px',
    zIndex:       '9999',
    background:   tipo === 'ok' ? '#1B3A6B' : '#DC2626',
    color:        '#fff',
    padding:      '12px 20px',
    borderRadius: '8px',
    fontSize:     '13.5px',
    fontWeight:   '600',
    boxShadow:    '0 4px 20px rgba(0,0,0,.2)',
    transition:   'opacity .3s',
    fontFamily:   "'Inter', sans-serif",
    maxWidth:     '320px'
  });

  document.body.appendChild(toast);
  setTimeout(() => {
    toast.style.opacity = '0';
    setTimeout(() => toast.remove(), 350);
  }, 3000);
}

// ================================================
//   UTILITÁRIOS
// ================================================
function esc(str) {
  return (str || '')
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;');
}
